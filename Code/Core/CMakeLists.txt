cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(SegelMacherCore VERSION 0.1.0 LANGUAGES CXX)
  include(CTest)
endif()
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/nlohmann/json.hpp")
  message(FATAL_ERROR "Missing vendored nlohmann/json 3.11.3. See ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/README.md.")
endif()
add_library(segelmacher_core STATIC src/core.cpp)
target_compile_features(segelmacher_core PUBLIC cxx_std_17)
target_include_directories(segelmacher_core PUBLIC include PRIVATE ../third_party)
if(MSVC)
  target_compile_options(segelmacher_core PRIVATE /utf-8 /W4)
endif()
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_include_directories(core_tests PRIVATE ../third_party)
  target_link_libraries(core_tests PRIVATE segelmacher_core)
  add_test(NAME core_contract COMMAND core_tests)
endif()
