cmake_minimum_required(VERSION 3.20)
project(SegelMacher VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
add_subdirectory(Code/Core)
add_executable(SegelMacher Code/CLI/main.cpp Code/CLI/job.cpp Code/CLI/opencl_backend.cpp)
target_link_libraries(SegelMacher PRIVATE segelmacher_core)
target_include_directories(SegelMacher PRIVATE Code/third_party Code/CLI)
if(WIN32)
  target_link_libraries(SegelMacher PRIVATE bcrypt)
endif()
if(MSVC)
  target_compile_options(SegelMacher PRIVATE /utf-8 /W4)
endif()
if(BUILD_TESTING)
  add_executable(cli_tests Code/tests/cli_tests.cpp Code/CLI/job.cpp Code/CLI/opencl_backend.cpp)
  target_include_directories(cli_tests PRIVATE Code/third_party Code/CLI)
  target_compile_definitions(cli_tests PRIVATE TEST_SOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(cli_tests PRIVATE segelmacher_core)
  if(WIN32)
    target_link_libraries(cli_tests PRIVATE bcrypt)
  endif()
  add_test(NAME cli_contract COMMAND cli_tests)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME cli_integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/Code/tests/integration.py" --exe "$<TARGET_FILE:SegelMacher>")
    set_tests_properties(cli_integration PROPERTIES TIMEOUT 1800)
  endif()
endif()
