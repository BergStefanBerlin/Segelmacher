cmake_minimum_required(VERSION 3.20)
project(SegelMacher VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
add_subdirectory(Code/Core)
find_package(Arrow CONFIG REQUIRED)
find_package(Parquet CONFIG REQUIRED)
set(SEGELPHYSIK_CORE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../Segelphysik/CORE/Code"
    CACHE PATH "SegelPhysik Core source directory providing modell_fingerprints.hpp")
if(NOT EXISTS "${SEGELPHYSIK_CORE_SOURCE_DIR}/include/segelphysik/modell_fingerprints.hpp")
  message(FATAL_ERROR "Set SEGELPHYSIK_CORE_SOURCE_DIR to the SegelPhysik Core Code directory")
endif()
set(core_library_default "")
if(WIN32 AND EXISTS "C:/spdeps/guibuild/segelphysik-core/Release/segelphysik_core.lib")
  set(core_library_default "C:/spdeps/guibuild/segelphysik-core/Release/segelphysik_core.lib")
endif()
set(SEGELPHYSIK_CORE_LIBRARY "${core_library_default}" CACHE FILEPATH
    "Existing Release Core library; empty builds the configured Core sources")
if(SEGELPHYSIK_CORE_LIBRARY)
  if(NOT EXISTS "${SEGELPHYSIK_CORE_LIBRARY}")
    message(FATAL_ERROR "SEGELPHYSIK_CORE_LIBRARY does not exist")
  endif()
  add_library(segelphysik_core STATIC IMPORTED)
  set_target_properties(segelphysik_core PROPERTIES
      IMPORTED_CONFIGURATIONS RELEASE
      IMPORTED_LOCATION_RELEASE "${SEGELPHYSIK_CORE_LIBRARY}"
      INTERFACE_INCLUDE_DIRECTORIES "${SEGELPHYSIK_CORE_SOURCE_DIR}/include"
      INTERFACE_COMPILE_FEATURES cxx_std_17)
  message(STATUS "Fingerprint Core: ${SEGELPHYSIK_CORE_LIBRARY} (Release)")
else()
  function(add_fingerprint_core)
    set(BUILD_TESTING OFF)
    add_subdirectory("${SEGELPHYSIK_CORE_SOURCE_DIR}" fingerprint-core EXCLUDE_FROM_ALL)
  endfunction()
  add_fingerprint_core()
  message(STATUS "Fingerprint Core: ${SEGELPHYSIK_CORE_SOURCE_DIR}")
endif()
add_executable(SegelMacher Code/CLI/main.cpp Code/CLI/job.cpp Code/CLI/opencl_backend.cpp)
target_link_libraries(SegelMacher PRIVATE segelmacher_core segelphysik_core Arrow::arrow_shared Parquet::parquet_shared)
target_include_directories(SegelMacher PRIVATE Code/third_party Code/CLI)
if(WIN32)
  target_link_libraries(SegelMacher PRIVATE bcrypt)
endif()
if(MSVC)
  target_compile_options(SegelMacher PRIVATE /utf-8 /W4)
endif()
if(BUILD_TESTING)
  add_executable(cli_tests Code/tests/cli_tests.cpp Code/CLI/job.cpp Code/CLI/opencl_backend.cpp)
  target_include_directories(cli_tests PRIVATE Code/third_party Code/CLI)
  target_compile_definitions(cli_tests PRIVATE TEST_SOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(cli_tests PRIVATE segelmacher_core segelphysik_core Arrow::arrow_shared Parquet::parquet_shared)
  if(WIN32)
    target_link_libraries(cli_tests PRIVATE bcrypt)
  endif()
  add_test(NAME cli_contract COMMAND cli_tests)
  add_executable(parquet_reader_tests Code/tests/parquet_reader.cpp)
  target_compile_features(parquet_reader_tests PRIVATE cxx_std_17)
  target_link_libraries(parquet_reader_tests PRIVATE Arrow::arrow_shared Parquet::parquet_shared)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME cli_integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/Code/tests/integration.py" --exe "$<TARGET_FILE:SegelMacher>" --parquet-reader "$<TARGET_FILE:parquet_reader_tests>" --model-schemas "${SEGELPHYSIK_CORE_SOURCE_DIR}/../../Modelle")
    set_tests_properties(cli_integration PROPERTIES TIMEOUT 1800)
  endif()
endif()
